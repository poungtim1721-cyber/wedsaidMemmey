ภาษาในช่องความหมาย: คำศัพท์และคันจิจากชุดเดิมส่วนใหญ่เป็นภาษาอังกฤษ ตรวจและแปลความหมายใน CSV ก่อนนำเข้า หากต้องการให้หน้าเว็บแสดงภาษาไทย

จำนวนคันจิจำแนกตามฟิลด์ JLPT ใหม่ในไฟล์คันจิเดิม จำนวนนี้ไม่ใช่การรับรองรายการตามหลักสูตรหรือชุดข้อสอบอย่างเป็นทางการ

2. ขั้นตอนตั้งค่าและนำเข้า

  1. เปิด https://supabase.com/dashboard/project/hrhhbahxwbolhvhrbhjn/sql แล้วเลือก SQL Editor
  2. หากยังไม่มีตาราง JLPT ให้รัน supabase/schema.sql; หากสร้างตารางคำศัพท์/คันจิไว้แล้ว ให้รันเฉพาะ supabase/manga-schema.sql
  3. ที่โฟลเดอร์โปรเจกต์ รัน .\supabase\export-content-csv.ps1 ใน PowerShell เพื่อสร้าง CSV ใน supabase\import
  4. ใน Table Editor นำเข้า jlpt_vocabulary.csv → jlpt_vocabulary และ jlpt_kanji.csv → jlpt_kanji
  5. รัน supabase/manga-schema.sql เพื่อสร้างตารางคำแปล พร้อมเพิ่มคำแปล 2 รายการที่มีอยู่แล้วในชุด manga-kana
  6. เพิ่มคำแปลอื่นใน manga_translations โดยระบุชุดมังงะ บท หน้า และลำดับบรรทัดให้ตรงกัน
  7. เพิ่มรายการไวยากรณ์ที่ตรวจสอบแล้วลง jlpt_grammar เมื่อมีข้อมูลพร้อม
  8. ตั้งค่า publishable key ใน data/supabase-client.js แล้วทดสอบเว็บไซต์ผ่านเว็บเซิร์ฟเวอร์

อย่านำเข้า id หรือ created_at จาก CSV ฐานข้อมูลจะสร้างค่าให้เอง ส่วนระดับให้ใช้ N5, N4, N3, N2 หรือ N1 เท่านั้น

3. SQL สำหรับรันใน Supabase (ส่วนที่ 1/2)

create table if not exists public.jlpt_vocabulary (
  id bigint generated by default as identity primary key,
  level text not null check (level in ('N5', 'N4', 'N3', 'N2', 'N1')),
  word text not null,
  reading text not null default '',
  meaning text not null default '',
  created_at timestamptz not null default now(),
  unique (level, word, reading)
);

create table if not exists public.jlpt_kanji (
  id bigint generated by default as identity primary key,
  level text not null check (level in ('N5', 'N4', 'N3', 'N2', 'N1')),
  character text not null,
  readings_on text[] not null default '{}',
  readings_kun text[] not null default '{}',
  meaning text not null default '',
  created_at timestamptz not null default now(),
  unique (level, character)
);

create table if not exists public.jlpt_grammar (
  id bigint generated by default as identity primary key,
  level text not null check (level in ('N5', 'N4', 'N3', 'N2', 'N1')),
  pattern text not null,
  reading text not null default '',
  meaning text not null default '',
  explanation text not null default '',
  example_jp text not null default '',
  example_th text not null default '',
  sort_order integer not null default 0,
  created_at timestamptz not null default now(),
  unique (level, pattern)
);

create table if not exists public.manga_translations (
  id bigint generated by default as identity primary key,
  series text not null check (series in ('manga-jp', 'manga-kana', 'manga-th')),
  chapter_number integer not null check (chapter_number > 0),
  page_number integer not null check (page_number > 0),
  line_number integer not null check (line_number > 0),
  romaji text not null default '',
  japanese text not null default '',
  meaning text not null default '',
  created_at timestamptz not null default now(),
  unique (series, chapter_number, page_number, line_number)
);

4. SQL สำหรับรันใน Supabase (ส่วนที่ 2/2)

create index if not exists jlpt_vocabulary_level_idx
  on public.jlpt_vocabulary (level, word);
create index if not exists jlpt_kanji_level_idx
  on public.jlpt_kanji (level, character);
create index if not exists jlpt_grammar_level_order_idx
  on public.jlpt_grammar (level, sort_order, id);
create index if not exists manga_translations_series_chapter_page_idx
  on public.manga_translations (series, chapter_number, page_number, line_number);

alter table public.jlpt_vocabulary enable row level security;
alter table public.jlpt_kanji enable row level security;
alter table public.jlpt_grammar enable row level security;
alter table public.manga_translations enable row level security;

drop policy if exists "Public can read JLPT vocabulary" on public.jlpt_vocabulary;
create policy "Public can read JLPT vocabulary"
  on public.jlpt_vocabulary for select to anon, authenticated
  using (true);

drop policy if exists "Public can read JLPT kanji" on public.jlpt_kanji;
create policy "Public can read JLPT kanji"
  on public.jlpt_kanji for select to anon, authenticated
  using (true);

drop policy if exists "Public can read JLPT grammar" on public.jlpt_grammar;
create policy "Public can read JLPT grammar"
  on public.jlpt_grammar for select to anon, authenticated
  using (true);

drop policy if exists "Public can read manga translations" on public.manga_translations;
create policy "Public can read manga translations"
  on public.manga_translations for select to anon, authenticated
  using (true);

grant select on public.jlpt_vocabulary to anon, authenticated;
grant select on public.jlpt_kanji to anon, authenticated;
grant select on public.jlpt_grammar to anon, authenticated;
grant select on public.manga_translations to anon, authenticated;

RLS เปิดใช้งานทั้งสี่ตาราง และอนุญาตให้อ่านเท่านั้น ไม่เปิดสิทธิ์เพิ่ม แก้ไข หรือลบจากหน้าเว็บ คำแปลเดิมสองรายการใช้ series manga-kana

ตารางคำแปลมีคอลัมน์ series, chapter_number, page_number, line_number, romaji, japanese และ meaning

5. เชื่อม API key อย่างปลอดภัย

  1. ใน Supabase เปิด Project Settings → API แล้วคัดลอก publishable key หรือ anon key แบบเดิม
  2. เปิด data/supabase-client.js ใน VS Code และตรวจว่าใส่ publishable/anon key ของโปรเจกต์ถูกต้องแล้ว ห้ามใส่ secret key
  3. บันทึกไฟล์ และทดสอบเว็บจาก local web server หรือโฮสต์เว็บไซต์ของคุณ ไม่ใช่เปิด HTML ด้วย file://
ห้ามใส่ service_role key ใน HTML, JavaScript หรือ PDF ที่เผยแพร่ต่อ เพราะ key นี้ข้าม RLS และให้สิทธิ์ระดับผู้ดูแล Publishable/anon key ใช้ใน browser ได้เมื่อเปิด RLS และกำหนด policy อย่างเหมาะสม

7. ตรวจสอบเมื่อทำเสร็จ

  • SQL Editor รันสำเร็จ และเห็นตารางทั้งสามใน Table Editor
